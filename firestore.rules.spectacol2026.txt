// Reguli Firestore pentru colecția spectacol2026_scans.
// Lipește blocul `match /spectacol2026_scans/...` în regulile existente din
// Firebase Console -> Firestore Database -> Rules (păstrează regulile pentru qr_scans etc.).

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Oricine (vizitatorul care scanează) poate DOAR să adauge o scanare.
    // Doar conturile @osfiir.ro logate pot citi statisticile.
    match /spectacol2026_scans/{scanId} {
      allow create: if request.resource.data.source is string
                    && request.resource.data.source.size() <= 40;
      allow read: if request.auth != null
                  && request.auth.token.email_verified == true
                  && request.auth.token.email.matches('.*@osfiir[.]ro$');
      allow update, delete: if false;
    }
  }
}
